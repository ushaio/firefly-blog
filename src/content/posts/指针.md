```go
// 交换a,b
func swap(a int, b int) {
	var temp int
	temp = a
	a = b
	b = temp
}

func main() {
	var a int = 10
	var b int = 20
	swap(a, b)
	fmt.Printf("a=%d ,b=%d", a, b)
}
----------
a=10 ,b=20
```

```go
func swap(a *int, b *int) {
	var temp int
	temp = *a // temp = main::a | temp=10
	*a = *b   // main::a = main::b | a=20
	*b = temp // main::b = temp | b=10
}
-----------
a=20 ,b=10
```

```go
var p *int
var q int
p = &a // 等于的是a的地址
q = a
fmt.Println("&a=", &a)
fmt.Println("p=", p)
fmt.Println("q=", q)
----------------
&a= 0xc00000a088
p= 0xc00000a088
q= 20
```

## 二级指针

```go
var p *int
p = &a // 等于的是a的地址
fmt.Println("&a=", &a)
fmt.Println("p=", p) // p=a的地址

var pp **int // 二级指针
pp = &p      // pp=p的地址
fmt.Println("&p=", &p)
fmt.Println("pp=", pp)
-------------
&p= 0xc000068040
pp= 0xc000068040
```
