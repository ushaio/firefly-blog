# 继承

父类Human

```go
type Human struct {
	name string
	sex  string
}

func (this *Human) Eat() {
	fmt.Println("Human.Eat()...")
}
func (this *Human) Walk() {
	fmt.Println("Human.Walk()...")
}
```

子类SuperMan

```go
// Human子类
type SuperMan struct {
	Human // SuperMan继承Human
	level int
}

// 重写父类方法Eat
func (this *SuperMan) Eat() {
	fmt.Println("SuperMan.Eat()...")
}

// 子类特有方法
func (this *SuperMan) Fly() {
	fmt.Println("SuperMan.Fly()...")
}

```

```go
func main() {
	h := Human{"zhangsan", "man"}
	h.Eat()
	h.Walk()

	//s := SuperMan{Human{"zhangsan", "man"}, 100}
	s := SuperMan{h, 100}
	s.level = 999
	s.Eat()
	s.Fly()
	s.Walk()
	fmt.Println("superman:", s)
}
```
