# 5.判断

```go
func main() {
	fmt.Print("输入年龄: ")
	n, err := fmt.Scan(&age) // 类型校验
	if err != nil {
		fmt.Println(n, err)
		return
	}
	fmt.Printf("你的年龄是： %d\n", age)
	
	// 中断式 卫语句 满足条件就结束
	if age <= 0 {
		fmt.Println("你还未出生")
		return
	}
	if age < 18 {
		fmt.Println("你还未成年")
		return
	}
	if age >= 18 && age <= 35 {
		fmt.Println("你还是青年")
		return
	}
	fmt.Println("你已中年")
}
```

```go
// 嵌套式
if age < 18 {
	if age <= 0 {
		fmt.Println("未出生")
	} else {
		fmt.Println("未成年")
	}
} else {
	if age <= 35 {
		fmt.Println("青年")
	} else {
		fmt.Println("中年")
	}
}
```

```go
// 多条件式
if age <= 0 {
	fmt.Println("未出生")
}
if age > 0 && age <= 18 {
	fmt.Println("未成年")
}
if age > 18 && age <= 35 {
	fmt.Println("青年")
}
```
