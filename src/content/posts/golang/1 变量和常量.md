# 1.变量和常量

## 变量

```go
var global_name1 = "global_name1"
var global_age = 123

// 全局变量必须要有var
// global_age1:= 29

func main() {
	// 先声明再赋值
	var name string
	name = "shai"
	fmt.Println(name)

	// 声明并赋值
	var name1 string = "shaiii"
	fmt.Println(name1)

	// 省略类型，自动判断
	var name2 = "SHAI"
	fmt.Println(name2)

	// 声明并赋值 短声明
	name3 := "SHAAAAI"
	fmt.Println(name3)

	// 定义多个变量
	var a1, a2 = 1, 2
	fmt.Println(a1, a2, a1+a2)

	// 多行多变量声明
	var (
		s1 string = "s1"
		s2 string = "s2"
		i1 int = 123
	)
	fmt.Println(s1, s2)

	// 调用函数
	hello()
	fmt.Println()

	// 使用 version.go 中的常量
	fmt.Println("version:", version.Version)
	fmt.Println("author:", version.Author)
}

// 定义函数
func hello() {
	fmt.Println("Func Hello")
	fmt.Println(global_name1)
	global_name1 = "edit: global_name1"
	fmt.Println(global_name1)
}
```

## 常量

不可修改

- 定义枚举类型
- 全局常量

`ioat`只能在`const(…)`中使用

```go
// 定义
const length int = 10
const (
	//BEIJING = 0
	//SHANGHAI = 1
	// 可以在const添加iota每行的ioat自动加，首行默认值为0
	BEIJING = iota // 0
	SHANGHAI // 1
	SHENZHEN // 2
)

fmt.Println(BEIJING, SHANGHAI, SHENZHEN) // 0 1 2
```

```go
const (
	CN = 10 * iota
	USA
	EN
)
fmt.Println(CN, USA, EN) // 0 10 20
```

```go
const (
	i1, i2 = iota + 1, iota + 2 // 1,2
	j1, j2                      //
	k1, k2

	u1, u2 = iota * 2, iota * 3 // ioat=3 u1=6,u2=9
	o1, o2                      // ioat=4 o1=4*2,o2=4*3
	p1, p2                      // ioat=5 p1=5*2,p2=5*3
)
```
